# Date Picker

A calendar for choosing a single date or a date range, with min/max and disabled-date rules.

**Stable** · Controls · [Guide on voltui.dev](https://voltui.dev/docs/components/date-picker)

> Recommended for production use. The public API follows the 1.x compatibility promise.

## When to use

- Booking, scheduling and filtering by date.

## Installation

Install the package and import the parts you use (the full example below also shows services and types):

```bash
npm install @voltui/components
```

```ts
import {
  VoltDatePickerCell,
  VoltDatePickerDateButton,
  VoltDatePickerGrid,
  VoltDatePickerLabel,
  VoltDatePickerNextMonth,
  VoltDatePickerPreviousMonth,
  VoltDatePicker,
  VoltDateRangePicker,
} from '@voltui/components';
```

Or copy the source into your project and own it: `npx @voltui/cli add date-picker`.

## Full example

The same code as the [voltui.dev guide](https://voltui.dev/docs/components/date-picker), as you would write it in an application:

````ts
import { Component, computed, signal } from '@angular/core';
import { FormControl, ReactiveFormsModule } from '@angular/forms';
import {
  NgpDatePickerCellRender,
  NgpDatePickerRowRender,
  VoltDatePicker,
  VoltDatePickerCell,
  VoltDatePickerDateButton,
  VoltDatePickerGrid,
  VoltDatePickerLabel,
  VoltDatePickerNextMonth,
  VoltDatePickerPreviousMonth,
} from '@voltui/components';

@Component({
  imports: [
    ReactiveFormsModule,
    VoltDatePicker,
    VoltDatePickerCell,
    VoltDatePickerDateButton,
    VoltDatePickerGrid,
    VoltDatePickerLabel,
    VoltDatePickerNextMonth,
    VoltDatePickerPreviousMonth,
    NgpDatePickerCellRender,
    NgpDatePickerRowRender,
  ],
  template: `
    <volt-date-picker [formControl]="date" [(focusedDate)]="focusedDate">
      <div class="flex items-center justify-between">
        <volt-date-picker-previous-month aria-label="Previous month">
          ‹
        </volt-date-picker-previous-month>
        <volt-date-picker-label>{{ label() }}</volt-date-picker-label>
        <volt-date-picker-next-month aria-label="Next month">
          ›
        </volt-date-picker-next-month>
      </div>

      <volt-date-picker-grid>
        <div class="grid grid-cols-7 gap-1 text-center text-xs text-muted-foreground">
          @for (day of weekDays; track day) {
            <span class="py-1">{{ day }}</span>
          }
        </div>
        <div *ngpDatePickerRowRender class="grid grid-cols-7 gap-1">
          <volt-date-picker-cell *ngpDatePickerCellRender="let day">
            <volt-date-picker-date-button>
              {{ day.getDate() }}
            </volt-date-picker-date-button>
          </volt-date-picker-cell>
        </div>
      </volt-date-picker-grid>
    </volt-date-picker>
  `,
})
export class MyComponent {
  date = new FormControl<Date | null>(new Date());
  focusedDate = signal(new Date());
  weekDays = ['Su', 'Mo', 'Tu', 'We', 'Th', 'Fr', 'Sa'];
  label = computed(() =>
    this.focusedDate().toLocaleString('en-US', {
      month: 'long',
      year: 'numeric',
    })
  );
}
````
