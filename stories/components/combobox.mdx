# Combobox

A text input with a filterable popup list. Supports local filtering, manual (server-side) filtering, loading state and multiple selection.

**Stable** · Forms & Inputs · [Guide on voltui.dev](https://voltui.dev/docs/components/combobox)

> Recommended for production use. The public API follows the 1.x compatibility promise.

## When to use

- Picking from long lists where typing narrows the options.
- Set `filterMode="manual"` and drive `items` yourself for server-side search.

## Installation

Install the package and import the parts you use (the full example below also shows services and types):

```bash
npm install @voltui/components
```

```ts
import {
  VoltComboboxButton,
  VoltComboboxDropdown,
  VoltComboboxInput,
  VoltComboboxOption,
  VoltCombobox,
} from '@voltui/components';
```

Or copy the source into your project and own it: `npx @voltui/cli add combobox`.

## Full example

The same code as the [voltui.dev guide](https://voltui.dev/docs/components/combobox), as you would write it in an application:

````ts
import { Component, signal } from '@angular/core';
import { FormControl, ReactiveFormsModule } from '@angular/forms';
import { VoltCombobox } from '@voltui/components';

// 1. A list you already have: Volt narrows it by label as the user types.
@Component({
  imports: [ReactiveFormsModule, VoltCombobox],
  template: `
    <volt-combobox [formControl]="framework" [items]="frameworks">
      <ng-template let-item>{{ item }}</ng-template>
    </volt-combobox>
  `,
})
export class LocalExample {
  framework = new FormControl<string | null>(null);
  frameworks = ['Angular', 'React', 'Vue', 'Svelte'];
}

interface Person {
  id: number;
  name: string;
}

declare function searchPeople(query: string): Promise<Person[]>; // your own API call

// 2. A server decides what matches: filterMode="manual" renders `items` exactly as supplied,
// (queryChange) reports what the user typed, and your code owns the request. Volt never
// fetches, debounces or cancels. Never combine server results with the default local filter.
@Component({
  imports: [ReactiveFormsModule, VoltCombobox],
  template: `
    <volt-combobox
      label="Owner"
      filterMode="manual"
      [items]="results()"
      [loading]="loading()"
      loadingText="Searching…"
      [emptyText]="'No one matches ' + query()"
      [itemLabel]="label"
      [compareWith]="sameId"
      [trackByFn]="byId"
      [formControl]="owner"
      (queryChange)="search($event)"
    />
  `,
})
export class RemoteExample {
  owner = new FormControl<Person | null>(null);
  results = signal<Person[]>([]);
  loading = signal(false);
  query = signal('');

  label = (person: unknown) => (person as Person).name;
  sameId = (a: unknown, b: unknown) => (a as Person | null)?.id === (b as Person | null)?.id;
  byId = (_index: number, person: unknown) => (person as Person).id;

  private latest = 0;

  async search(query: string) {
    this.query.set(query);
    const mine = ++this.latest; // latest wins: a slow answer to "a" must not replace the one to "al"
    this.loading.set(true);
    try {
      const people = await searchPeople(query);
      if (mine === this.latest) this.results.set(people);
    } finally {
      if (mine === this.latest) this.loading.set(false);
    }
  }
}
````
